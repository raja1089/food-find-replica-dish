## ✅ Complete Node.js Code (Replit API)

```js
// server.js or routes/cook.js
const express = require('express');
const mysql = require('mysql2/promise');
const cors = require('cors');
require('dotenv').config();

const app = express();
app.use(cors());
app.use(express.json());

// Connect to DB
const db = mysql.createPool({
  host: process.env.DB_HOST,
  user: process.env.DB_USER,
  password: process.env.DB_PASSWORD,
  database: process.env.DB_NAME,
});

// POST /api/cook/register
app.post('/api/cook/register', async (req, res) => {
  try {
    const {
      firstName, lastName, email, phone,
      kitchenName, kitchenType, address, city, state, pincode,
      experience, description, cuisineTypes,
      fssaiLicense, gstNumber, panNumber,
      latitude, longitude
    } = req.body;

    // Validate minimum required fields
    if (!firstName || !phone || !kitchenName || !address || !cuisineTypes?.length) {
      return res.status(400).json({ error: 'Missing required fields' });
    }

    const conn = await db.getConnection();

    // Check if user already exists
    const [existingUser] = await conn.query(
      'SELECT id FROM users WHERE mobile_number = ? OR email = ?',
      [phone, email]
    );

    let userId;
    if (existingUser.length > 0) {
      userId = existingUser[0].id;
    } else {
      // Insert new user
      const [userResult] = await conn.query(
        `INSERT INTO users (first_name, last_name, email, mobile_number, user_type, role_id, status, is_verified, latitude, longitude, created_at)
         VALUES (?, ?, ?, ?, 'cook', 2, 1, 1, ?, ?, NOW())`,
        [firstName, lastName, email, phone, latitude || null, longitude || null]
      );
      userId = userResult.insertId;
    }

    // Insert into cooks table
    await conn.query(
      `INSERT INTO cooks 
       (user_id, name, address, experience_years, cuisine_id, availability, created_at, updated_at, landmark, instagram, twitter, facebook, profile_image_url, latitude, longitude)
       VALUES (?, ?, ?, ?, ?, 1, NOW(), NOW(), '', '', '', '', '', ?, ?)`,
      [userId, kitchenName, address, experience || 0, null, latitude || null, longitude || null]
    );

    // Optional: Save business docs in another table
    await conn.query(
      `INSERT INTO cook_documents (cook_id, fssai_number, gst_number, pan_number, created_at)
       VALUES (?, ?, ?, ?, NOW())`,
      [userId, fssaiLicense || '', gstNumber || '', panNumber || '']
    );

    conn.release();

    res.json({ success: true, message: 'Cook registered successfully', user_id: userId });
  } catch (err) {
    console.error('Registration Error:', err.message);
    res.status(500).json({ error: 'Internal Server Error' });
  }
});

// Start server
const PORT = process.env.PORT || 5000;
app.listen(PORT, () => console.log(`Server running on port ${PORT}`));
```
